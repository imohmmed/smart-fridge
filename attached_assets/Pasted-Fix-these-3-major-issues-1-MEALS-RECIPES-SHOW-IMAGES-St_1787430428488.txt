    'تبولة': 'tabbouleh', 'فتوش': 'fattoush',
    'بيتزا': 'pizza', 'برغر': 'burger',
    'شوكولاتة': 'chocolate dessert',
    'كيك': 'cake', 'حليب': 'milk glass'
  };
  
  const english = Object.keys(translations).find(k => 
    arabicName.includes(k)
  );
  const query = english ? translations[english] : 'food dish';
  
  return `https://source.unsplash.com/300x200/?${query},food`;
}

// Notification types to generate automatically:
function checkAndGenerateNotifications() {
  const notifications = [];
  const fridgeItems = getUserFridgeItems();
  const today = new Date();
  
  fridgeItems.forEach(item => {
    // Expiry warning - 3 days before
    if (item.expiryDate) {
      const expiry = new Date(item.expiryDate);
      const daysLeft = Math.ceil(
        (expiry - today) / (1000 * 60 * 60 * 24)
      );
      
      if (daysLeft <= 0) {
        notifications.push({
          id: `expired_${item.id}`,
          type: 'danger',
          icon: '⚠️',
          title: 'طعام منتهي الصلاحية',
          message: `${item.name} انتهت صلاحيته!`,
          time: 'الآن'
        });
      } else if (daysLeft <= 3) {
        notifications.push({
          id: `expiring_${item.id}`,
          type: 'warning', 
          icon: '🕐',
          title: 'قريب انتهاء الصلاحية',
          message: `${item.name} ينتهي خلال ${daysLeft} أيام`,
          time: `${daysLeft} أيام`
        });
      }
    }
    
    // Low quantity warning
    if (item.quantity <= 1) {
      notifications.push({
        id: `low_${item.id}`,
        type: 'info',
        icon: '📦',
        title: 'كمية منخفضة',
        message: `${item.name} كميته منخفضة، أضفه لقائمة التسوق`,
        time: 'اليوم'
      });
    }
  });
  
  // Daily calorie goal
  const consumed = getTodayCalories();
  const goal = getUserCalorieGoal();
  if (consumed >= goal) {
    notifications.push({
      id: 'calorie_goal',
      type: 'success',
      icon: '🎯',
      title: 'وصلت هدفك!',
      message: `أكملت ${goal} سعرة حرارية اليوم`,
      time: 'اليوم'
    });
  }
  
  // Water reminder
  const cups = getTodayWaterCups();
  if (cups < 4) {
    notifications.push({
      id: 'water_reminder',
      type: 'info',
      icon: '💧',
      title: 'تذكير شرب الماء',
      message: `شربت ${cups} أكواب فقط، هدفك 8 أكواب`,
      time: 'اليوم'
    });
  }
  
  return notifications;
}
